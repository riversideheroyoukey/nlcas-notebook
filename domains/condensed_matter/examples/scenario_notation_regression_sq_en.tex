Notation regression (second quantization). How a prime reads, and three operand spellings.

define c as an electron annihilation operator

\[
  c
\]

define a as a photon annihilation operator

\[
  a
\]

a'(k)

\begin{equation}
  a^{\dagger}(k)
  \label{eq:E1}
\end{equation}

G(t, t')

\begin{equation}
  G(t, t')
  \label{eq:E2}
\end{equation}

$\sigma'$

\begin{equation}
  \sigma'
  \label{eq:E3}
\end{equation}

$H = a'(k) a(k)$

\begin{equation}
  H = a^{\dagger}(k) a(k)
  \label{eq:E4}
\end{equation}

evolve c(k,u) under E4

\begin{equation}
  c_{ku}
  \label{eq:E5}
\end{equation}

evolve c(1, u) under E4

\begin{equation}
  c_{1u}
  \label{eq:E6}
\end{equation}

evolve $c_{1\uparrow}$ under E4

\begin{equation}
  c_{1\uparrow}
  \label{eq:E7}
\end{equation}

$P = \sum_\sigma a^\dagger_{1\sigma} a_{2\sigma}$

\begin{equation}
  P = \sum_\sigma a^\dagger_{1\sigma} a_{2\sigma}
  \label{eq:E8}
\end{equation}

$Q = \sum_\sigma a^\dagger_{1\sigma} a_{1\sigma}$

\begin{equation}
  Q = \sum_\sigma a^\dagger_{1\sigma} a_{1\sigma}
  \label{eq:E9}
\end{equation}

the commutator of P and Q

\begin{equation}
  - {a}_{1{,+}}^{\dagger} {a}_{2{,+}} - {a}_{1{,-}}^{\dagger} {a}_{2{,-}}
  \label{eq:E10}
\end{equation}

the commutator of $a_{1,+}$ and $a^\dagger_{1,+}$

\begin{equation}
  1
  \label{eq:E11}
\end{equation}

the commutator of $a_{1+}$ and $a^\dagger_{1-}$

\begin{equation}
  0
  \label{eq:E12}
\end{equation}

$\lim_{x \to 0} \frac{\sin x}{x}$

\begin{equation}
  \lim_{x \to 0} \frac{\sin x}{x}
  \label{eq:E13}
\end{equation}
